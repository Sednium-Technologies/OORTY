// Native GGUF Inference Streaming via llama.cpp Kotlin Bindings
class LlamaHelper(private val context: Context, private val uri: Uri) {
    suspend fun generateStream(
        prompt: String,
        systemInstruction: String,
        temperature: Float = 0.7f
    ): Flow<String> = flow {
        val model = LlamaModel(context, uri)
        val session = model.createSession()
        
        session.streamTokens(prompt, systemInstruction, temperature).collect { chunk ->
            emit(chunk) // Real-time token emission back to Jetpack Compose UI
        }
    }.flowOn(Dispatchers.Default)
}

// Zero-GPU TF-IDF Title Generation in <10ms
object LiteRtTitleGen {
    fun generateTitle(userMessage: String, modelResponse: String): String {
        val tokens = (userMessage + " " + modelResponse)
            .lowercase()
            .split(Regex("[^a-zA-Z0-9]+"))
            .filter { it.length > 3 && it !in StopWords.SET }
            
        val topKeywords = tokens.groupingBy { it }.eachCount()
            .entries.sortedByDescending { it.value }
            .take(4).map { it.key.replaceFirstChar { c -> c.uppercase() } }
            
        return if (topKeywords.isNotEmpty()) topKeywords.joinToString(" ") else "New Chat"
    }
}

object StorageHelper {
    suspend fun saveChatToVault(context: Context, session: ChatSession) = withContext(Dispatchers.IO) {
        val vaultDir = File(
            Environment.getExternalStoragePublicDirectory(Environment.DIRECTORY_DOCUMENTS),

    }

HELP & KNOWLEDGE BASE

Frequently Asked Questions

Everything you need to know about on-device inference, Obsidian sync, and API keys.

How does on-device GGUF inference work on Android without root?

Oorty runs an optimized ARM NEON llama.cpp runtime embedded natively via JNI inside the APK. You can download quantized GGUF models directly via the built-in HuggingFace Hub or load any GGUF file from your phone storage via Android's Storage Access Framework (SAF). Everything processes 100% locally in your device's memory.

How do I connect Oorty directly to the Obsidian Mobile app?

Oorty automatically saves every conversation into Documents/Oorty/chats/ with complete YAML frontmatter (tags, timestamps, token counts). In the Obsidian Android app, tap "Open folder as vault", select Documents/Oorty/, and your entire chat history will instantly appear as searchable Markdown notes with graph links!

What is the Dynamic RAM Watchdog and how does it prevent crashes?

Loading large neural networks on Android can trigger the Linux Out-Of-Memory (OOM) killer. Oorty uses HardwareChecker.kt to inspect real-time available RAM before loading. If a model exceeds 55% of free RAM, it issues a helpful warning. If it exceeds 85%, it suggests a lighter quantization (e.g. Q4_K_M) to guarantee zero system freezes.

Can I use cloud AI models alongside local GGUF models?

Yes! Oorty is a complete multi-provider orchestrator. You can switch between on-device GGUF and cloud providers (Google Gemini 2.5, Anthropic Claude 3.7 Sonnet, OpenAI GPT-4o, Groq LPU, NVIDIA NIM, xAI Grok, OpenRouter) simply by inputting your personal API keys (BYOK) with zero intermediary server fees.

How do I run a background llama-server via Termux?

Install Termux on your device, build llama.cpp using pkg install clang cmake git && make -j4, and launch ./llama-server -m ~/model.gguf --port 8080. In Oorty Settings, select LOCAL SERVER and set the Base URL to http://localhost:8080/v1!
