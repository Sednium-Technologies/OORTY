// Native Coroutine Flow Stream from llama.cpp JNI
fun generateStream(prompt: String, params: InferenceParams): Flow<StreamChunk> = callbackFlow {
    val startTime = System.nanoTime()
    var tokenCount = 0
    
    llamaEngine.streamTokens(prompt) { token ->
        tokenCount++
        val elapsedSec = (System.nanoTime() - startTime) / 1_000_000_000.0
        val tokPerSec = if (elapsedSec > 0) tokenCount / elapsedSec else 0.0
        trySend(StreamChunk(text = token, tokPerSec = tokPerSec))
    }
    awaitClose { llamaEngine.cancel() }
}.flowOn(Dispatchers.Default)

// Sub-10ms Title Extraction without loading LLM
fun extractTopicTitle(firstMessage: String): String {
    val clean = firstMessage.replace(PUNCTUATION_REGEX, "")
    val keywords = clean.split(" ")
        .filter { it.length > 3 && !STOP_WORDS.contains(it.lowercase()) }
        .take(4)
        .joinToString(" ") { it.replaceFirstChar { c -> c.uppercase() } }
    return keywords.ifEmpty { "New Conversation" }
}
app/api/StreamingThoughtParser.kt • Dynamic <think> Parser 
// Real-time parse of DeepSeek <think> reasoning tags
class StreamingThoughtParser {
    fun parseChunk(chunk: String): ParsedThoughtState {
        val thinkStart = chunk.indexOf("<think>")
        val thinkEnd = chunk.indexOf("</think>")
        
        return when {
            thinkStart != -1 && thinkEnd != -1 ->
                ParsedThoughtState.Complete(thought = chunk.substring(thinkStart + 7, thinkEnd))
            thinkStart != -1 ->
                ParsedThoughtState.Thinking(partial = chunk.substring(thinkStart + 7))
            else -> ParsedThoughtState.Output(response = chunk)
        }
    }
}

FREQUENTLY ASKED QUESTIONS

Everything You Need To Know

Clear answers to your technical and architectural questions.

What is Oorty and why is it unique?
Oorty is a native Android application engineered in Kotlin with Jetpack Compose. It allows you to run local open-source GGUF AI models directly on your phone's processor using llamacpp-kotlin, with zero internet requirement, while automatically syncing every turn as an open Markdown document into an Obsidian vault.
Which GGUF models are supported on my phone?
Any GGUF quantization format (Q4_K_M, Q5_K_M, Q8_0) is supported. For phones with 4GB RAM, we recommend Qwen 2.5 (0.5B); for 6GB–8GB RAM, Llama 3.2 (1B) or Gemma 2 (2B); and for 12GB+ devices, Phi-3 Mini (3.8B) or Llama 3.1 (8B).
How does the Obsidian vault dual-write work?
Whenever you chat, Oorty persists the session to SQLite for instantaneous UI queries and simultaneously writes an open .md file with structured YAML frontmatter (session ID, model name, tokens/sec, and timestamp) directly into Documents/Oorty/chats/.
Can I use cloud frontier models as well?
Yes. Oorty features a built-in Multi-Cloud Router allowing you to bring your own API keys (BYOK) for Google Gemini 2.5, Anthropic Claude 3.7, OpenAI, Groq, Mistral, and NVIDIA NIM. Your keys are encrypted locally on-device.
Does Oorty require root access or Termux?
No. Oorty is a self-contained, native APK package that ships with pre-compiled ARM64 JNI binaries for llama.cpp. Simply install the APK on Android 8.0+ and start running models immediately.
